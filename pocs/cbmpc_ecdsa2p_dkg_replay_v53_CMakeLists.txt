cmake_minimum_required(VERSION 3.16)
project(cbmpc-ecdsa2p-dkg-replay-v53 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED CBMPC_REPO_ROOT)
  message(FATAL_ERROR "CBMPC_REPO_ROOT is required")
endif()
if(NOT DEFINED CBMPC_SOURCE_DIR)
  set(CBMPC_SOURCE_DIR "${CBMPC_REPO_ROOT}/build/install/public")
endif()
include(${CBMPC_REPO_ROOT}/cmake/macros.cmake)
include(${CBMPC_REPO_ROOT}/cmake/arch.cmake)
include(${CBMPC_REPO_ROOT}/cmake/openssl.cmake)
include(${CBMPC_REPO_ROOT}/cmake/compilation_flags.cmake)
set(CBMPC_LIB_DIR "${CBMPC_SOURCE_DIR}/lib")
if(EXISTS "${CBMPC_SOURCE_DIR}/lib/Release/libcbmpc.a")
  set(CBMPC_LIB_DIR "${CBMPC_SOURCE_DIR}/lib/Release")
endif()
add_executable(cbmpc-ecdsa2p-dkg-replay-v53 cbmpc_ecdsa2p_dkg_replay_v53.cpp)
target_include_directories(cbmpc-ecdsa2p-dkg-replay-v53 PRIVATE ${CBMPC_SOURCE_DIR}/include)
target_link_directories(cbmpc-ecdsa2p-dkg-replay-v53 PRIVATE ${CBMPC_LIB_DIR})
target_link_libraries(cbmpc-ecdsa2p-dkg-replay-v53 PRIVATE cbmpc pthread)
link_openssl(cbmpc-ecdsa2p-dkg-replay-v53)
