cmake_minimum_required(VERSION 3.16)
project(cbmpc-ecdsa2p-transport-v45 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED CBMPC_SOURCE_DIR)
  message(FATAL_ERROR "CBMPC_SOURCE_DIR is required")
endif()
set(CBMPC_LIB_DIR "${CBMPC_SOURCE_DIR}/lib")
if(EXISTS "${CBMPC_SOURCE_DIR}/lib/Debug/libcbmpc.a")
  set(CBMPC_LIB_DIR "${CBMPC_SOURCE_DIR}/lib/Debug")
endif()
add_executable(cbmpc-ecdsa2p-transport-v45 cbmpc_ecdsa2p_transport_mutation_v45.cpp)
target_include_directories(cbmpc-ecdsa2p-transport-v45 PRIVATE ${CBMPC_SOURCE_DIR}/include)
target_link_directories(cbmpc-ecdsa2p-transport-v45 PRIVATE ${CBMPC_LIB_DIR})
target_compile_options(cbmpc-ecdsa2p-transport-v45 PRIVATE -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined -fno-sanitize=enum)
target_link_options(cbmpc-ecdsa2p-transport-v45 PRIVATE -fsanitize=address,undefined)
target_link_libraries(cbmpc-ecdsa2p-transport-v45 PRIVATE cbmpc pthread ssl crypto)
