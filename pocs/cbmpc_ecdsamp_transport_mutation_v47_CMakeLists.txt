cmake_minimum_required(VERSION 3.16)
project(cbmpc-ecdsamp-transport-v47 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED CBMPC_REPO_ROOT)
  message(FATAL_ERROR "CBMPC_REPO_ROOT is required")
endif()
if(NOT DEFINED CBMPC_SOURCE_DIR)
  message(FATAL_ERROR "CBMPC_SOURCE_DIR is required")
endif()
include(${CBMPC_REPO_ROOT}/cmake/macros.cmake)
include(${CBMPC_REPO_ROOT}/cmake/arch.cmake)
include(${CBMPC_REPO_ROOT}/cmake/openssl.cmake)
set(CBMPC_LIB_DIR "${CBMPC_SOURCE_DIR}/lib")
if(EXISTS "${CBMPC_SOURCE_DIR}/lib/Debug/libcbmpc.a")
  set(CBMPC_LIB_DIR "${CBMPC_SOURCE_DIR}/lib/Debug")
endif()
add_executable(cbmpc-ecdsamp-transport-v47 cbmpc_ecdsamp_transport_mutation_v47.cpp)
target_include_directories(cbmpc-ecdsamp-transport-v47 PRIVATE ${CBMPC_SOURCE_DIR}/include)
target_link_directories(cbmpc-ecdsamp-transport-v47 PRIVATE ${CBMPC_LIB_DIR})
target_compile_options(cbmpc-ecdsamp-transport-v47 PRIVATE -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined -fno-sanitize=enum -Wno-c++11-narrowing)
target_link_options(cbmpc-ecdsamp-transport-v47 PRIVATE -fsanitize=address,undefined)
target_link_libraries(cbmpc-ecdsamp-transport-v47 PRIVATE cbmpc pthread)
link_openssl(cbmpc-ecdsamp-transport-v47)
