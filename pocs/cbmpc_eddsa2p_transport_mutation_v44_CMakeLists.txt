cmake_minimum_required(VERSION 3.16)
project(cbmpc-eddsa2p-transport-v44 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED CBMPC_SOURCE_DIR)
  message(FATAL_ERROR "CBMPC_SOURCE_DIR is required")
endif()
set(CBMPC_LIB_DIR "${CBMPC_SOURCE_DIR}/lib")
if(EXISTS "${CBMPC_SOURCE_DIR}/lib/Debug/libcbmpc.a")
  set(CBMPC_LIB_DIR "${CBMPC_SOURCE_DIR}/lib/Debug")
endif()
add_executable(cbmpc-eddsa2p-transport-v44 cbmpc_eddsa2p_transport_mutation_v44.cpp)
target_include_directories(cbmpc-eddsa2p-transport-v44 PRIVATE ${CBMPC_SOURCE_DIR}/include)
target_link_directories(cbmpc-eddsa2p-transport-v44 PRIVATE ${CBMPC_LIB_DIR})
target_compile_options(cbmpc-eddsa2p-transport-v44 PRIVATE -O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined -fno-sanitize=enum -Wno-c++11-narrowing)
target_link_options(cbmpc-eddsa2p-transport-v44 PRIVATE -fsanitize=address,undefined)
target_link_libraries(cbmpc-eddsa2p-transport-v44 PRIVATE cbmpc pthread ssl crypto)
