cmake_minimum_required(VERSION 3.16)
project(cbmpc-eddsa-ac-policy-v55 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED CBMPC_REPO_ROOT)
  message(FATAL_ERROR "CBMPC_REPO_ROOT is required")
endif()
if(NOT DEFINED CBMPC_SOURCE_DIR)
  set(CBMPC_SOURCE_DIR "${CBMPC_REPO_ROOT}/build/install/public")
endif()
include(${CBMPC_REPO_ROOT}/cmake/macros.cmake)
include(${CBMPC_REPO_ROOT}/cmake/arch.cmake)
include(${CBMPC_REPO_ROOT}/cmake/openssl.cmake)
include(${CBMPC_REPO_ROOT}/cmake/compilation_flags.cmake)
set(CBMPC_LIB_DIR "${CBMPC_SOURCE_DIR}/lib")
if(EXISTS "${CBMPC_SOURCE_DIR}/lib/Release/libcbmpc.a")
  set(CBMPC_LIB_DIR "${CBMPC_SOURCE_DIR}/lib/Release")
endif()
add_executable(cbmpc-eddsa-ac-policy-v55 cbmpc_eddsa_ac_policy_substitution_v55.cpp)
target_include_directories(cbmpc-eddsa-ac-policy-v55 PRIVATE ${CBMPC_SOURCE_DIR}/include)
target_link_directories(cbmpc-eddsa-ac-policy-v55 PRIVATE ${CBMPC_LIB_DIR})
target_link_libraries(cbmpc-eddsa-ac-policy-v55 PRIVATE cbmpc pthread)
link_openssl(cbmpc-eddsa-ac-policy-v55)
